def generate_sentence(template):
    """Điền dữ liệu thực tế vào mẫu câu để tạo ra câu thoại có ý nghĩa hoàn chỉnh."""
    return template.format(
        book=random.choice(book_titles),
        genre=random.choice(genres),
        order_code=random.choice(order_codes),
        city=random.choice(cities),
        customer=random.choice(customer_titles),
        price=random.choice(price_strings),
        time=random.choice(time_strings),
        cancel_reason=random.choice(cancellation_reasons)
    )

def build_corpus(target_count=TARGET_COUNT, output_path=OUTPUT_FILE):
    """Tạo ra tập câu thoại đa dạng, chuẩn Tiếng Việt và giàu ý nghĩa nghiệp vụ."""
    all_templates = (
        order_lookup_templates * 4 +
        order_cancel_templates * 3 +
        book_search_templates * 4 +
        drm_reader_templates * 3 +
        support_templates * 3 +
        conversational_templates * 2
    )

    sentences = set()

    # Sinh câu từ các mẫu hội thoại với các thực thể thay đổi
    while len(sentences) < target_count:
        tpl = random.choice(all_templates)
        sentence = generate_sentence(tpl).strip()
        sentences.add(sentence)

    sentence_list = list(sentences)[:target_count]
    random.shuffle(sentence_list)

    # Ghi tệp
    dir_name = os.path.dirname(output_path)
    if dir_name:
        os.makedirs(dir_name, exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:
        for s in sentence_list:
            f.write(s + "\n")

    print(f"Da tao thanh cong {len(sentence_list)} cau thoai Chatbot CSKH thuan Viet tai: {output_path}")
    return sentence_list

if __name__ == "__main__":
    build_corpus(TARGET_COUNT, OUTPUT_FILE)
